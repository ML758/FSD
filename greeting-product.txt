import React from "react"; 

function App() { 

const date = new Date(); 
const hour = date.getHours(); 
const time = date.toLocaleTimeString();
 
let greeting = ""; 

if (hour < 12) { 

greeting = "Good Morning!"; 

} else if (hour < 18) { 

greeting = "Good Afternoon!"; 

} else { 

greeting = "Good Evening!"; 

} 

return ( 
<div>
<h1>{greeting}</h1> 
<p>Current Time: {time}</p>

</div>

); 

} 

export default App;
