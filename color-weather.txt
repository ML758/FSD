App.js- 
import React, { useState } from 'react';
function App() {
  const [color, setColor] = useState('lightblue');

  function changeColor() {
    const random = '#' + Math.floor(Math.random()*16777215).toString(16);
    setColor(random);
  }
return (
    <div
      onClick={changeColor}
      style={{ width: 200, height: 200, background: color}}>
    </div>
  );
}
export default App;
