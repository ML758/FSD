App.js- 

import React, { useState } from "react"; 

 

function App() { 

const [count, setCount] = useState(0); 

 

return ( 

<div style={{ textAlign: "center", marginTop: "50px" }}> 

<h2>Value: {count}</h2> 

<button onClick={() => setCount(count - 1)}>-</button> 

<button onClick={() => setCount(count + 1)}>+</button> 

</div> 

); 

} 

 

export default App;
